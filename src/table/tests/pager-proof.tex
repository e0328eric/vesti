\def\vestiTBmeasurepage#1#2#3#4{%
  \testfirst=#2\relax \testlast=#3\relax
  \testcount=\testlast \advance\testcount by-\testfirst \advance\testcount by1
  \vestiTBpagecost=25pt \multiply\vestiTBpagecost by\testcount
  \advance\vestiTBpagecost by\csname test#1\endcsname
  \advance\vestiTBpagecost by\csname test#4\endcsname\relax}
\def\testrows{%
  \ifnum\testscan>\testlast\let\testrownext\relax\else
    \vbox to25pt{\vfil\hbox{ROW \the\testscan}\vfil}%
    \global\advance\testnext by1 \advance\testscan by1
    \let\testrownext\testrows
  \fi\testrownext}
\def\vestiTBrenderpage#1#2#3#4{%
  \vestiTBmeasurepage{#1}{#2}{#3}{#4}%
  \ifnum\testfirst=\testnext\else\errmessage{Rows duplicated or skipped}\fi
  \advance\testpages by1
  \immediate\write16{PAGER FRAGMENT \the\testpages: #1 / #2--#3 / #4}%
  \ifcase\testpages\or
    \ifnum\testlast=3\else\errmessage{Forced boundary was skipped}\fi
  \or
    \ifnum\testlast=7\else\errmessage{Largest legal prefix was not selected}\fi
  \or
    \ifnum\testlast=10\else\errmessage{Final fragment wrong}\fi
    \edef\testactualfoot{#4}\def\testexpectedfoot{lastfoot}%
    \ifx\testactualfoot\testexpectedfoot\else\errmessage{Final footer missing}\fi
  \else\errmessage{Too many fragments}\fi
  \setbox\vestiTBcanvas=\vbox to\vestiTBpagecost{%
    \baselineskip=0pt\lineskip=0pt\lineskiplimit=16383pt
    \vbox to\csname test#1\endcsname{\vfil\hbox{#1}\vfil}%
    \testscan=\testfirst \testrows
    \vbox to\csname test#4\endcsname{\vfil\hbox{#4}\vfil}}%
  \testheight=\ht\vestiTBcanvas
  \setbox\vestiTBcanvas=\hbox{\lower\testheight\box\vestiTBcanvas}%
  \ht\vestiTBcanvas=0pt \dp\vestiTBcanvas=\testheight}
